# ====================================================================
# 代理检查
# 作用：确认 git/curl 使用的 7890 代理是否在真实本机环境可连
# ====================================================================
git config --global --get http.proxy
git config --global --get https.proxy
nc -vz 127.0.0.1 7890
curl -I --proxy http://127.0.0.1:7890 https://github.com --connect-timeout 5

# ====================================================================
# AprilTag 本地构建验证
# 作用：指定当前 Python、pybind11 和 OpenCV_DIR，检查 OpenCV 链接组件是否收敛
# ====================================================================
cmake -S cv/apriltag_detection -B /tmp/cpp_tools_apriltag_build -DPYTHON_EXECUTABLE=$(which python3) -Dpybind11_DIR=$(python3 -m pybind11 --cmakedir) -DOpenCV_DIR=/usr/local/lib/cmake/opencv4
cmake --build /tmp/cpp_tools_apriltag_build --parallel 8
otool -L cv/apriltag_detection/lib/apriltag_detection*.so   # macOS
ldd cv/apriltag_detection/lib/apriltag_detection*.so        # Ubuntu/Linux
